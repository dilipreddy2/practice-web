// Experiment: Fetch data from an API using Fetch API.
load.onclick=async()=>{out.textContent='Loading...';try{let r=await fetch('https://jsonplaceholder.typicode.com/todos/1');let d=await r.json();out.textContent=JSON.stringify(d,null,2)}catch(e){out.textContent='Request failed: '+e.message}};
